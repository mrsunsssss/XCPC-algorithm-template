> 树形dp

https://ac.nowcoder.com/acm/contest/95334/I   好题，很深刻的推导和dp定义

自底向上dp，记 $cnt_i$ 表示以节点 $i$ 为根的连通块（树）个数。设当前算节点 $u$ ，而节点 $v$ 是 $u$ 的儿子。于是 $cnt_u = cnt_u + cnt_u cnt_v$ . 后半部分表示钦定选择了 $v$ 节点作为根的树的数量，故还要加上没有选择的 $v$ 节点的部分即前半部分。即 $cnt_u = (cnt_v+1)cnt_u$ . 这个要放到最后更新。

记录 $dp_i$ 表示以节点 $i$ 为根的树中的答案，这里的答案暂时先不考虑顺序，最后输出再乘2。

首先在$u$ 或 $v$ 树中找两个点，然后合并答案。

 $dp_u = dp_u + dp_u cnt_v + dp_v cnt_u$ ，第二个表示合并前的 $u$ 子树中找两个点，钦定合并了 $v$ 为根的一颗树形成的连通块；第三个表示 $v$ 子树找两个点，并且使得它们与 $u$ 合并；第一个表示合并前的 $u$ 子树中找两个点，并且不与 $v$ 子树做操作，即先前的答案，它不会凭空消失。也可看作是 $dp_u = dp_u(cnt_v+1) + dp_v cnt_u$ 。

其次需要在 $u$ 和 $v$ 两颗树中各找一个点，也将它算入 $dp_u$ 的答案中，这里也要同时考虑同样的两个点在不同连通块的贡献也要记入。记 $f_i$ 和 $g_i$ 分别表示 以 $i$ 为根的所有树中所有点的 $0$ 和 $1$ 的贡献，同样的点在不同连通块的贡献也要记入。

$dp_u = dp_u + f_u g_v +f_vg_u$

$f_u = f_u + f_ucnt_v +f_vcnt_u = f_u(cnt_v + 1) + f_vcnt_u$

$g_u = g_u + g_ucnt_v +g_vcnt_u = g_u(cnt_v + 1) + g_vcnt_u$

跟上面的推导类似。然后排一下式子的顺序，就完成了。

```C++
 void Solve() {
    int n;cin >> n;
    vector<int> a(n + 1);for (int i = 1;i <= n;i++) cin >> a[i];
    vector<vector<int>> g(n + 1);
    for (int i = 1;i < n;i++) {
        int u, v;cin >> u >> v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    int res = 0;
    vector<int> dp(n + 1), cnt(n + 1);
    vector<array<int, 2>> f(n + 1);
    auto dfs = [&](auto&& dfs, int u, int p, int w)->void {
        cnt[u] = 1;
        f[u][a[u] >> w & 1] = 1;
        f[u][~a[u] >> w & 1] = 0;
        dp[u] = 0;
        for (auto v : g[u]) {
            if (v == p) continue;
            dfs(dfs, v, u, w);
            dp[u] = (dp[u] * (cnt[v] + 1) % MOD + dp[v] * cnt[u] % MOD) % MOD;
            dp[u] = (dp[u] + f[u][0] * f[v][1] % MOD + f[u][1] * f[v][0] % MOD) % MOD;
            f[u][0] = (f[u][0] * (1 + cnt[v]) % MOD + f[v][0] * cnt[u] % MOD) % MOD;
            f[u][1] = (f[u][1] * (1 + cnt[v]) % MOD + f[v][1] * cnt[u] % MOD) % MOD;
            cnt[u] = cnt[u] * (cnt[v] + 1) % MOD;
        }
        res = (res + dp[u] * (1 << w) % MOD) % MOD;
        };
    for (int i = 0;i <= 30;i++) {
        dfs(dfs, 1, 1, i);
    }
    cout << res * 2 % MOD << endl;
}
```
