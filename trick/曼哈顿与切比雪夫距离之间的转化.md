分别作出曼哈顿距离和切比雪夫距离为 $1$ 的图像，发现分别是一个长为 $\sqrt2$ 的旋转 $\frac{\pi}{4}$ 的正方形，一个长为 $2$ 的正方形。

于是利用坐标旋转公式，曼哈顿坐标旋转 $\frac{\pi}{4}$ 得到 $(\frac{\sqrt 2}{2} (x+y),\frac{\sqrt 2}{2} (x-y))$ .  然后再扩大 $\sqrt 2$ 倍即可得到切比雪夫的图像，即 $(x+y,x-y)$ 。同理，切比雪夫转化为曼哈顿距离，除以 $\sqrt 2$ 即可得到 $(\frac12 (x+y),\frac12(x-y))$ 。

void Solve() {
    int n, m;cin >> n >> m;
    vector<vector<int>> a(n + 1, vector<int>(m + 1));
    vector<vector<ll>> res(n + 1, vector<ll>(m + 1));
    vector<vector<array<int, 2>>> X(n + m + 1), Y(n + m + 1);
    ll Sx = 0, Sy = 0;
    for (int i = 1;i <= n;i++) {
        for (int j = 1;j <= m;j++) {
            cin >> a[i][j];
            int x = (i + j), y = (i - j + m);
            X[x].push_back({ i,j });
            Y[y].push_back({ i,j });
            Sx += x, Sy += y;
        }
    }
    int cnt = 0;ll sum = 0;
    for (int i = 0;i <= n + m;i++) {
        for (auto [x, y] : X[i]) {
            res[x][y] += 1LL * cnt * i - sum;
            res[x][y] += Sx - sum - 1LL * (n * m - cnt) * i;
        }
        cnt += X[i].size();
        sum += 1LL * X[i].size() * i;
    }
    cnt = 0, sum = 0;
    for (int i = 0;i <= n + m;i++) {
        for (auto [x, y] : Y[i]) {
            res[x][y] += 1LL * cnt * i - sum;
            res[x][y] += Sy - sum - 1LL * (n * m - cnt) * i;
        }
        cnt += Y[i].size();
        sum += 1LL * Y[i].size() * i;
    }
    for (int i = 1;i <= n;i++) {
        for (int j = 1;j <= m;j++) {
            cout << 1LL * res[i][j] * a[i][j] / 2;
            if (j != m) cout << " ";
        }
        cout << endl;
    }
}
