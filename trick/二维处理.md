#pragma GCC optmize("Ofast")

vector<vector<int>> transpose(const vector<vector<int>>& a) {//矩阵转置
    int n = a.size();
    int m = a[0].size();
    vector<vector<int>> b(m, vector<int>(n));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            b[j][i] = a[i][j];
        }
    }
    return b;
}
vector<vector<int>> rotate(const vector<vector<int>>& a) {//矩阵旋转
    int n = a.size();
    int m = a[0].size();
    vector<vector<int>> b(m, vector<int>(n));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            b[m - j - 1][i] = a[i][j];
        }
    }
    return b;
}

void Solve() {
    string s;cin >> s;
    int n = s.size();

    vector<vector<int>> a(n, vector<int>(n));
    for (int i = 0;i < n;i++) {
        for (int j = 0;j < n;j++) {
            a[i][j] = s[(j - i + n) % n] - '0';
        }
    }

    vector<vector<int>> b(n, vector<int>(n));
    for (int i = 0;i < n;i++) {
        for (int j = 0;j < n;j++) {
            if (a[i][j] == 0) {
                b[i][j] = (j == 0 ? 0 : b[i][j - 1]) + 1;
            }
        }
    }
    int res = 0;
    for (int j = 0;j < n;j++) {
        vector<int> up(n), down(n), st;
        for (int i = 0;i < n;i++) {
            while (st.size() and b[st.back()][j] > b[i][j]) {
                down[st.back()] = i;
                st.pop_back();
            }
            st.push_back(i);
        }
        while (st.size()) {
            down[st.back()] = n;
            st.pop_back();
        }
        for (int i = n - 1;i >= 0;i--) {
            while (st.size() and b[st.back()][j] > b[i][j]) {
                up[st.back()] = i;
                st.pop_back();
            }
            st.push_back(i);
        }
        while (st.size()) {
            up[st.back()] = -1;
            st.pop_back();
        }
        for (int i = 0;i < n;i++) {
            res = max(res, b[i][j] * (down[i] - up[i] - 1));
        }
    }
    for (int tc = 0;tc < 4;tc++) {
        for (int i = 0;i < n;i++) {
            for (int j = 0;j < n;j++) {
                if (a[i][j] == 0) {
                    b[i][j] = (j ? b[i][j - 1] : 0) + 1;
                }
                else {
                    b[i][j] = 0;
                }
            }
        }
        for (int j = 0;j < n;j++) {
            int lst = 0;
            for (int i = 0;i < n;i++) {
                if (b[i][j] > lst) lst++;
                else lst = b[i][j];
                res = max(res, (1 + lst) * lst / 2);
            }
        }
        a = rotate(a);
    }
    cout << res << endl;
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);cout.tie(0);
    int T = 1;
    //cin >> T;
    Prework();
    while (T--) Solve();
}
