多项式方法: O(nlogn)，模数为质数

$(1+x+x^2+...)(1+x^2+(x^2)^2+...)(1+x^3+(x^3)^2+...)...=\frac{1}{1-x} \frac{1}{1-x^2}\frac{1}{1-x^3}...$

$=\prod\limits_{i=1}\frac{1}{1-x^i} = e^{\sum\limits_{i=1}\ln\frac{1}{1-x^i}}=e^{\sum\limits_{i=1}\ln\frac{1}{1-x^i}}= e^{\sum\limits_{i=1}\sum\limits_{j=1}\frac{x^{ij}}{j}}$

使用后面的项数通过调和级数O(nlogn)获得，然后用多项式快速幂。

当模数为非质数时，此时由于需要计算逆元，不能再使用多项式做法。(并且该数字增长很快，如果算出具体的数字再取模，用中国剩余定理也是难以接受的。)

dp方法: O( $n\sqrt n$ ) , 如果要计算多项，需要算出dp2之后并进行求和，记为dp3，然后卷积dp1和dp3

```
int MOD;
void cadd(int& x, int y) {
    x += y;
    if (x >= MOD) x -= MOD;
}

void Solve() {
    int n;cin >> n >> MOD;
    int m = sqrt(n) + 1;
    vector<int> dp1(n + 1);//用了前i个数，和为j
    dp1[0] = 1;
    for (int i = 1;i <= m;i++) {
        for (int j = i;j <= n;j++) {
            cadd(dp1[j], dp1[j - i]);
        }
    }
    vector dp2(m + 1, vector<int>(n + 1));//用了i个大于m的数，和为j
    dp2[0][0] = 1;
    for (int i = 1;i <= m;i++) {
        for (int j = i;j <= n;j++) {
            cadd(dp2[i][j], dp2[i][j - i]);
            if (j >= m + 1) cadd(dp2[i][j], dp2[i - 1][j - m - 1]);
        }
    }
    int res = 0;
    for (int i = 0;i <= n;i++) {
        int sum = 0;
        for (int j = 0;j <= m;j++) {
            cadd(sum, dp2[j][n - i]);
        }
        res = (res + sum * dp1[i] % MOD) % MOD;
    }
    cout << res << endl;
}
```
