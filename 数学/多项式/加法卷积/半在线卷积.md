https://qoj.ac/contest/1846/problem/9042

每块可以分开计算。

设dp[i]表示长度为i的分割序列，所需要期望操作次数。

设f[i]表示长度为i的分割序列数量。 $f_i = i!-\sum\limits_{j=1}^{i-1}f_j\cdot (i-j)!$ .可以通过半在线卷积求出(分治FFT)

$dp_i = 1 + \sum\limits_{j=0}^{i}\sum\limits_{k=0}^{i-j}k!(i-j-k)!dp_j f_j$

首先观察到 $k!(i-j-k)!$ 是一个卷积的形式，把他看作是 $G(i)$.

观察到 $H = dp_jf_j$ 和 $G$ 也是一个卷积的形式，于是再次做半在线卷积就做完了。

```
void Solve() {
    int m;cin >> m;
    vector<int> a(m + 1);for (int i = 1;i <= m;i++) cin >> a[i];
    int mx = -1, cnt = 0, n = 1;
    vector<int> vec;
    for (int i = 1;i <= m;i++) {
        mx = max(mx, a[i]);
        cnt++;
        if (i == mx) {
            n = max(n, cnt);
            vec.push_back(cnt);
            mx = -1;
            cnt = 0;
        }
    }
    Poly G(n + 1);
    for (int i = 0;i <= n;i++) G[i] = fact[i];
    G = G * G;
    Poly F(n + 1), dp(n + 1);
    auto cdq1 = [&](auto&& cdq1, int l, int r) {
        if (l == r) {
            F[l] = Add(fact[l], F[l]);
            return;
        }
        int mid = l + r >> 1;
        cdq1(cdq1, l, mid);
        Poly A(mid - l + 1);for (int i = l;i <= mid;i++) A[i - l] = F[i];
        Poly B(r - l);for (int i = 1;i <= r - l;i++) B[i - 1] = fact[i];
        A = A * B;
        for (int i = mid + 1;i <= r;i++)  F[i] = Dec(F[i], A[i - l - 1]) % MOD;
        cdq1(cdq1, mid + 1, r);
        };
    cdq1(cdq1, 1, n);
    auto cdq2 = [&](auto&& cdq2, int l, int r) {
        if (l == r) {
            dp[l] = mul(Add(fact[l], dp[l]), inv(Dec(fact[l], F[l]) % MOD));
            return;
        }
        int mid = l + r >> 1;
        cdq2(cdq2, l, mid);
        Poly A(mid - l + 1);for (int i = l;i <= mid;i++) A[i - l] = mul(dp[i], F[i]);
        Poly B(r - l);for (int i = 1;i <= r - l;i++) B[i - 1] = G[i];
        A = A * B;
        for (int i = mid + 1;i <= r;i++)  dp[i] = Add(dp[i], A[i - l - 1]);
        cdq2(cdq2, mid + 1, r);
        };
    cdq2(cdq2, 1, n);
    int res = 0;
    for (auto i : vec) {
        res = Add(res, dp[i]);
    }
    cout << res << endl;
}
```
