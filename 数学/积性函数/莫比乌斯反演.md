莫比乌斯反演

莫比乌斯反演因数形式 , $f(n) = \sum\limits_{d|n} g(d)\Leftrightarrow g(n) = \sum\limits_{d|n} f(d)\mu(\frac{n}{d})$.

莫比乌斯反演倍数形式 , $g(n) = \sum\limits_{n|d} f(d) \Leftrightarrow f(n) = \sum\limits_{n|d} g(d)\mu(\frac{d}{n})$

证明：考虑逆推这个式子,  $\sum\limits_{n|d}\mu(\frac{d}{n})g(d) = \sum\limits_{k=1}\mu(k)g(kn)=\sum\limits_{k=1}\mu(k)\sum\limits_{kn|D}f(D) = \sum\limits_{n|D}f(D)\sum\limits_{k|\frac{D}{n}}\mu(k)=\sum\limits_{n|D}f(D)[n=D]=f(n)$

莫比乌斯函数

$\sum\limits_{d|n}\mu(d) = [n=1]$ , 用 $f(n)$ 替换 $n$ ，即可得到 $[f(n)=1]=\sum\limits_{d|f(n)}\mu(d)$


欧拉函数

$\sum\limits_{d|n} \varphi(d) = n$ , 用 $f(n)$ 替换 $n$ ，即可得到 $f(n)= \sum\limits_{d|f(n)}\varphi(d)$

技巧1： $\sum\limits_{i=1}^{m} [k|i] f(i) = \sum\limits_{i=1}^{\lfloor\frac{m}{k}\rfloor} f(ik)$   

技巧2： $\sum\limits_{i=1}^{n}\sum\limits_{j=1}^{m}\sum\limits_{d|gcd(i,j)} f(i,j,d) = \sum\limits_{d=1}^{n}\sum\limits_{i=1}^{n}\sum\limits_{j=1}^{m}[d|i][d|j] f(i,j,d) = \sum\limits_{d=1}^{n}\sum\limits_{i=1}^{\lfloor\frac{n}{d}\rfloor}\sum\limits_{j=1}^{\lfloor\frac{m}{d}\rfloor} f(di,dj,d)$

技巧3： $\sum\limits_{i=1}^{n}\sum\limits_{j=1}^{m} f(gcd(i,j))\cdot g(i,j) =  \sum\limits_{d=1}^n \sum\limits_{i=1}^{n}\sum\limits_{j=1}^{m}  f(d)\cdot g(i,j)\cdot [d=\gcd(i,j)] $

$= \sum\limits_{d=1}^{n}f(d) \sum\limits_{i=1}^{\lfloor\frac{n}{d}\rfloor}\sum\limits_{j=1}^{\lfloor\frac{m}{d}\rfloor}  g(di,dj) [\gcd(i,j) = 1] =\sum\limits_{d=1}^nf(d) \sum\limits_{i=1}^{\lfloor\frac{n}{d}\rfloor}\sum\limits_{j=1}^{\lfloor\frac{m}{d}\rfloor}   g(di,dj) \sum\limits_{D|gcd(i,j)}\mu(D) $

$=\sum\limits_{d=1}^nf(d)\sum\limits_{D=1}^{\lfloor\frac{n}{d}\rfloor}\mu(D) \sum\limits_{i=1}^{\lfloor\frac{n}{dD}\rfloor}\sum\limits_{j=1}^{\lfloor\frac{m}{dD}\rfloor}  g(dDi,dDj) $

 $=\sum\limits_{k=1}^{n}\sum\limits_{i=1}^{\lfloor\frac{n}{k}\rfloor}\sum\limits_{j=1}^{\lfloor\frac{m}{k}\rfloor}  g(ki,kj) \sum\limits_{d|k}f(d)\mu(\frac{k}{d}) $

做法1：莫比乌斯反演

$dp_i=\sum\limits_{j=1}^{i-1}dp_j-\sum\limits_{j=1}^{i-1}dp_j[gcd(a_i,a_j)=1]$

$=\sum\limits_{j=1}^{i-1}dp_j-\sum\limits_{j=1}^{i-1}dp_j\sum\limits_{d|gcd(a_i,a_j)}\mu(d)=\sum\limits_{j=1}^{i-1}dp_j-\sum\limits_{d|a_i}\mu(d)\sum\limits_{j=1}^{i-1}dp_j[d|a_j]$

视作 $f(d)=\sum\limits_{j=1}^{i-1}dp_j[d|a_j]$ ，动态更新即可。时间复杂度 $O(n\sum\limits_{1\le i\le n} d(a_i))$.



做法2：取每个数的质因子，更新的时候取出若干个不同的质因子，进行容斥加入贡献。具体的，如果取出奇数个，则符号为正。为偶数，符号为负。

由于不同的质因子个数在该数据范围下最多为 $8$ 个，最终的时间复杂度为 $O(n2^8\cdot8)$

$\sum\limits_{i=1}^{n}\sum\limits_{j=1}^{n}\frac{i+j}{gcd(i,j)} = \sum\limits_{d=1}^{n}\sum\limits_{i=1}^{n}\sum\limits_{j=1}^{n} \frac{i+j}{d[gcd(i,j) =d]} =\sum\limits_{d=1}^{n}\sum\limits_{i=1}^{\lfloor\frac{n}{d}\rfloor}\sum\limits_{j=1}^{\lfloor\frac{n}{d}\rfloor} \frac{d(i+j)}{d[gcd(di,dj) =d]}  = \sum\limits_{d=1}^{n}\sum\limits_{i=1}^{\lfloor\frac{n}{d}\rfloor}\sum\limits_{j=1}^{\lfloor\frac{n}{d}\rfloor} (i+j)[gcd(i,j)=1]$

$=\sum\limits_{d=1}^{n}\sum\limits_{i=1}^{\lfloor\frac{n}{d}\rfloor}\sum\limits_{j=1}^{\lfloor\frac{n}{d}\rfloor} (i+j)\sum\limits_{D|gcd(i,j)} \mu(D)=\sum\limits_{d=1}^{n}\sum\limits_{i=1}^{\lfloor\frac{n}{d}\rfloor}\sum\limits_{j=1}^{\lfloor\frac{n}{d}\rfloor} (i+j)\sum\limits_{D=1}^{n} \mu(D)[D|gcd(i,j)] $

$=\sum\limits_{d=1}^{n}\sum\limits_{i=1}^{\lfloor\frac{n}{dD}\rfloor}\sum\limits_{j=1}^{\lfloor\frac{n}{dD}\rfloor} D(i+j)\sum\limits_{D=1}^{n} \mu(D)=\sum\limits_{D=1}^{n}D\mu(D)\sum\limits_{d=1}^{n}\sum\limits_{i=1}^{\lfloor\frac{n}{dD}\rfloor}\sum\limits_{j=1}^{\lfloor\frac{n}{dD}\rfloor} (i+j) $

$=\sum\limits_{d=1}^{n}\sum\limits_{D=1}^{n}D\mu(D)\sum\limits_{i=1}^{\lfloor\frac{n}{dD}\rfloor}\sum\limits_{j=1}^{\lfloor\frac{n}{dD}\rfloor} (i+j)$

$=\sum\limits_{k=1}^{n}\sum\limits_{d|k}d\mu(d)\sum\limits_{i=1}^{\lfloor\frac{n}{k}\rfloor}\sum\limits_{j=1}^{\lfloor\frac{n}{k}\rfloor} (i+j)$  

$=\sum\limits_{k=1}^{n}f(k)g(\lfloor\frac{n}{k}\rfloor)$



其中 $g(n)=\sum\limits_{i=1}^{n}\sum\limits_{j=1}^{n}(i+j) =\sum\limits_{i=1}^{n}\sum\limits_{j=1}^{n}i+\sum\limits_{i=1}^{n}\sum\limits_{j=1}^{n}j= 2\sum\limits_{i=1}^{n}ni = n^2(n+1)$



记上面要计算的为 $x$ .题目所求即为 $\frac{x-2n}{2}$ 。



然后有两种做法：

1.预处理 $f$ 前缀和，每组询问整除分块。复杂度 $O(NlogN + \sum \sqrt n)$

2.预处理枚举 $k$ ，维护要计算结果的差分。每次给 $res[k*j]$ 加上 $f[i]\cdot (g(j)-g(j-1))$ .最后前缀和即可。 复杂度 $O(NlogN + T)$


```C++
#pragma GCC optimize("Ofast")
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
//#define int long long 
#define endl '\n'

const int MOD = 1e9 + 7;
const int inv2 = MOD + 1 >> 1;
const int N = 1e6 + 10;
int mu[N];
int np[N];vector<int> p;
int f[N];
int res[N];
int g(int n) {
    return 1LL * n * n % MOD * (n + 1) % MOD;
}
void cadd(int& x, int y) {
    x += y;
    if (x >= MOD) x -= MOD;
}
void Prework(int n = N - 10) {
    np[0] = np[1] = 1;mu[1] = 1;
    for (int i = 2;i <= n;i++) {
        if (!np[i]) {
            p.push_back(i);
            mu[i] = MOD - 1;
        }
        for (auto j : p) {
            if (i * j > n) break;
            np[i * j] = 1;
            if (i % j == 0) {
                break;
            }
            else {
                mu[i * j] = (MOD - mu[i]) % MOD;
            }
        }
    }
    for (int i = 1;i <= n;i++) {
        for (int j = i;j <= n;j += i) {
            cadd(f[j], 1LL * i * mu[i] % MOD);
        }
    }
    for (int i = 1;i <= n;i++) {
        for (int j = 1;j * i <= n;j++) {
            cadd(res[i * j], 1LL * f[i] * (g(j) - g(j - 1) + MOD) % MOD);
        }
    }
    for (int i = 1;i <= n;i++) cadd(res[i], res[i - 1]);
    for (int i = 1;i <= n;i++) res[i] = (1LL * res[i] * inv2 % MOD - i + MOD) % MOD;
}


void Solve() {
    int n;cin >> n;
    cout << res[n] << endl;
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);cout.tie(0);
    int T = 1;
    cin >> T;
    Prework();
    while (T--) Solve();
}
```
