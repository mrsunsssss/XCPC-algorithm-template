vector<vector<int>> transpose(const vector<vector<int>>& a) {//矩阵转置
    int n = a.size();
    int m = a[0].size();
    vector<vector<int>> b(m, vector<int>(n));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            b[j][i] = a[i][j];
        }
    }
    return b;
}
vector<vector<int>> rotate(const vector<vector<int>>& a) {//矩阵旋转
    int n = a.size();
    int m = a[0].size();
    vector<vector<int>> b(m, vector<int>(n));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            b[m - j - 1][i] = a[i][j];
        }
    }
    return b;
}

#include <bits/stdc++.h>

void run(vector<int>& vec) {
    int n = vec.size();
    int pos = 0;
    for (int i = 0; i < n; i++) {
        if (vec[i] != 0) {
            swap(vec[pos], vec[i]);
            pos++;
        }
    }
    for (int i = pos;i < n;i++) vec[i] = 0;
    for (int i = 0; i < n - 1; i++) {
        if (vec[i] != 0 and vec[i] == vec[i + 1]) {
            vec[i] *= 2;
            vec[i + 1] = 0;
            i++;
        }
    }
    pos = 0;
    for (int i = 0; i < n; i++) {
        if (vec[i] != 0) {
            swap(vec[pos], vec[i]);
            pos++;
        }
    }
    for (int i = pos;i < n;i++) vec[i] = 0;
}

void do_left(vector<vector<int>>& mat) {
    for (auto& r : mat) run(r);
}
void do_right(vector<vector<int>>& mat) {
    for (auto& r : mat) {
        reverse(r.begin(), r.end());
        run(r);
        reverse(r.begin(), r.end());
    }
}

void do_up(vector<vector<int>>& mat) {
    int n = mat.size(), m = mat[0].size();
    for (int j = 0; j < m; j++) {
        vector<int> col;
        for (int i = 0; i < n; i++) col.push_back(mat[i][j]);
        run(col);
        for (int i = 0; i < n; i++) mat[i][j] = col[i];
    }
}

void do_down(vector<vector<int>>& mat) {
    int n = mat.size(), m = mat[0].size();
    for (int j = 0; j < m; ++j) {
        vector<int> col;
        for (int i = 0; i < n; ++i) col.push_back(mat[i][j]);
        reverse(col.begin(), col.end());
        run(col);
        reverse(col.begin(), col.end());
        for (int i = 0; i < n; ++i) mat[i][j] = col[i];
    }
}
