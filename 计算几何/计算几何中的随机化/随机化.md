平面最近点对，考虑随即旋转坐标系后，将所有点按照x排序，计算相邻点对的最小值。

```C++
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
#define int long long 
#define endl '\n'

#define debugP(x) cerr<<#x<<":("<<x[0]<<','<<x[1]<<")"<<endl
using PI = array<int, 2>;
using VI = array<int, 2>;
using LI = array<PI, 2>;
using CI = vector<PI>;

using ld = long double;
using PD = array<ld, 2>;
using VD = array<ld, 2>;
using LD = array<PD, 2>;
using CD = vector<PD>;
const ld eps = 1e-9;

using ty_g = int;
using P = PI;
using V = VI;
using L = LI;
using C = CI;

const ld Pi = 3.14159265357;
VD rotate(VD l, ld angle) {
    ld cosa = cos(angle), sina = sin(angle);
    return { l[0] * cosa - l[1] * sina, l[0] * sina + l[1] * cosa };
}
mt19937 gen(random_device{}());
uniform_int_distribution<int> rd(0, 359);
ld dis(PD u, PD v) { return sqrt((u[0] - v[0]) * (u[0] - v[0]) + (u[1] - v[1]) * (u[1] - v[1])); }

void Prework() {

}
void Solve() {
    int n;cin >> n;
    vector<PD> a(n);for (int i = 0;i < n;i++) cin >> a[i][0] >> a[i][1];
    ld res = 1e18;
    for (int _ = 0;_ < 10;_++) {
        ld ang = Pi * rd(gen) / 180;
        for (auto& i : a) i = rotate(i, ang);
        sort(a.begin(), a.end());
        for (int i = 1;i < n;i++) res = min(res, dis(a[i], a[i - 1]));
    }
    cout << res << endl;
}

signed main() {
    ios::sync_with_stdio(0);
    cin.tie(0);cout.tie(0);
    cout << fixed << setprecision(4);
    int T = 1;
    //cin >> T;
    Prework();
    while (T--) Solve();
}
```
